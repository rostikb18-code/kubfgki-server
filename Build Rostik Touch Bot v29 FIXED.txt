# Rostik Touch Bot v30.2

v30.2 is the UI/vision/navigation revision based on the supplied Brawl Stars screenshots and the supplied gameplay video.

## App flow

- MainActivity: ИГРАТЬ / СТАТИСТИКА / РАЗРЕШЕНИЯ
- ИГРАТЬ starts the overlay and opens Brawl Stars after capture permission.
- Floating AI bubble stays over the game.
- AI panel: БОТ / ОЧЕРЕДЬ / РАЗМЕР.
- Queue: all 108 catalog entries are visible; selected fighters appear in the right queue with current/target cups.
- Size: overlay width/height are configurable and persisted.
- Statistics: total time, net cups, matches, wins/losses/draws and win rate.

## v30.2 vision/navigation changes

- Brawler selection uses visual reference matching plus OCR, with whole-grid low-resolution scanning instead of only a fixed 5x3 coordinate table.
- Brawler scrolling is verified with a frame fingerprint and reverses direction if the list does not change.
- Critical navigation does not tap an arbitrary mode. It searches specifically for Brawl Ball and only uses a coordinate fallback after a whole-screen reference match.
- The Brawl Ball lobby from the supplied video is included as a reference.
- Play and Play Again have OCR plus a constrained yellow-action visual fallback.
- Match entry requires repeated game-HUD evidence including both joystick regions; a single false player detection cannot immediately start the match.
- A missing player for a few frames is treated as temporary perception loss rather than a match end.
- Result cup deltas require repeated confirmation before statistics/queue progress are updated.
- The old hard-coded `nearest to 800` cup heuristic was removed; the configured current-cup value is used as the expected value.
- Gameplay dispatch uses detected joystick positions when available.

## Important limitation

The gameplay controller is still a deterministic reactive controller, not a trained neural policy. The vision/navigation and UI are hardened, but match win rate cannot be guaranteed from source code alone. The app deliberately stops instead of issuing a blind action when a critical screen cannot be confirmed.

The catalog contains 108 brawler IDs. The supplied screenshots provide visual assets for 91 of them; the remaining catalog entries are still selectable in the UI and use OCR/name matching until dedicated reference images are supplied.

package com.rostik.touchbot

import android.accessibilityservice.AccessibilityService
import android.accessibilityservice.GestureDescription
import android.graphics.Path
import android.view.accessibility.AccessibilityEvent

class BotAccessibilityService : AccessibilityService() {
    companion object { @Volatile var instance: BotAccessibilityService? = null }

    override fun onServiceConnected(){
        super.onServiceConnected()
        instance=this
    }
    override fun onAccessibilityEvent(event: AccessibilityEvent?) {}
    override fun onInterrupt(){}

    override fun onDestroy(){
        if(instance===this) instance=null
        super.onDestroy()
    }

    fun dispatch(frame:ControlFrame, left:Point, right:Point, duration:Long=80L):Boolean {
        val builder=GestureDescription.Builder()
        val lp=Path().apply{moveTo(left.x*resources.displayMetrics.widthPixels,left.y*resources.displayMetrics.heightPixels);lineTo(
            (left.x+frame.moveX*.12f).coerceIn(0f,1f)*resources.displayMetrics.widthPixels,
            (left.y+frame.moveY*.12f).coerceIn(0f,1f)*resources.displayMetrics.heightPixels)}
        builder.addStroke(GestureDescription.StrokeDescription(lp,0,duration))
        val rp=Path().apply{moveTo(right.x*resources.displayMetrics.widthPixels,right.y*resources.displayMetrics.heightPixels);lineTo(
            (right.x+frame.aimX*.14f).coerceIn(0f,1f)*resources.displayMetrics.widthPixels,
            (right.y+frame.aimY*.14f).coerceIn(0f,1f)*resources.displayMetrics.heightPixels)}
        if(frame.fire || kotlin.math.abs(frame.aimX)+kotlin.math.abs(frame.aimY)>.05f)
            builder.addStroke(GestureDescription.StrokeDescription(rp,0,duration))
        if(frame.useSuper && frame.superPoint!=null){
            val s=Path().apply{
                moveTo(frame.superPoint.x*resources.displayMetrics.widthPixels,frame.superPoint.y*resources.displayMetrics.heightPixels)
                lineTo(frame.superPoint.x*resources.displayMetrics.widthPixels,frame.superPoint.y*resources.displayMetrics.heightPixels)
            }
            builder.addStroke(GestureDescription.StrokeDescription(s,0,55L))
        }
        return dispatchGesture(builder.build(),null,null)
    }

    fun swipe(from: Point, to: Point, duration: Long = 420L): Boolean {
        val w = resources.displayMetrics.widthPixels
        val h = resources.displayMetrics.heightPixels
        val path = Path().apply {
            moveTo(from.x * w, from.y * h)
            lineTo(to.x * w, to.y * h)
        }
        return dispatchGesture(
            GestureDescription.Builder()
                .addStroke(GestureDescription.StrokeDescription(path, 0, duration))
                .build(), null, null
        )
    }

    fun tap(p:Point):Boolean{
        val x=p.x*resources.displayMetrics.widthPixels
        val y=p.y*resources.displayMetrics.heightPixels
        val path=Path().apply{moveTo(x,y);lineTo(x,y)}
        return dispatchGesture(GestureDescription.Builder().addStroke(
            GestureDescription.StrokeDescription(path,0,55L)).build(),null,null)
    }
}

package com.rostik.touchbot

import android.content.Context
import android.graphics.Bitmap

/** Process-local bridge shared by MainActivity, capture and floating overlay. */
object BotRuntime {
    private lateinit var app: Context
    val queue: QueueEngine = QueueEngine()
    val supervisor: MatchSupervisor by lazy { MatchSupervisor(queue, app.assets) }

    @Volatile var captureActive: Boolean = false
        private set

    fun init(context: Context) {
        app = context.applicationContext
        StatsStore.init(app)
    }

    fun submitFrame(bitmap: Bitmap) {
        // Ownership is transferred to MatchSupervisor. Do not keep another reference
        // here because the supervisor recycles each consumed frame.
        supervisor.submitLatestFrame(bitmap)
    }

    fun setCaptureActive(active: Boolean) { captureActive = active }
}

===== app/src/main/java/com/rostik/touchbot/BrawlerSelector.kt =====
package com.rostik.touchbot

import android.content.res.AssetManager
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.graphics.Color
import kotlin.math.abs
import kotlin.math.max
import kotlin.math.min

/**
 * Adaptive brawler-card matcher.
 *
 * The live screen is searched in two independent ways:
 *  - OCR identifies the card label when readable;
 *  - a low-resolution visual scan searches the whole brawler grid, so the
 *    selector is not tied to a fixed 5x3 coordinate table.
 *
 * The caller must still verify the tap on a subsequent frame.
 */
class BrawlerSelector(private val assets: AssetManager) {
    data class Target(val point: Point, val confidence: Float, val source: String)

    private val cache = HashMap<String, Bitmap?>()
    private var lastGridFingerprint = 0L
    private var lastScrollAt = 0L

    fun findByOcr(lines: List<OcrLine>, id: String): Target? {
        val aliases = BrawlerCatalog.aliases(id).map(::norm).filter { it.length >= 3 }
        val line = lines.asSequence()
            .filter { it.rect.t > .08f && it.rect.b < .95f }
            .mapNotNull { line ->
                val value = norm(line.text)
                if (aliases.any { a -> value == a || value.contains(a) || (a.contains(value) && value.length >= 4) }) line else null
            }
            .maxByOrNull { scoreTextLine(it, aliases) }
            ?: return null

        val cy = line.rect.center().y
        // Name labels are at the lower part of a card. Move to the card body,
        // but clamp aggressively so a two-line OCR box cannot jump to another row.
        val y = (cy - 0.075f).coerceIn(.13f, .90f)
        return Target(Point(line.rect.center().x.coerceIn(.03f, .97f), y), scoreTextLine(line, aliases), "OCR")
    }

    fun findByAvatar(bitmap: Bitmap, id: String): Target? {
        val template = loadTemplate(id) ?: return null
        val direct = scanWholeGrid(bitmap, template)
        if (direct != null) return direct
        return scanExpectedGrid(bitmap, template)
    }

    fun find(bitmap: Bitmap, lines: List<OcrLine>, id: String): Target? {
        val visual = findByAvatar(bitmap, id)
        val ocr = findByOcr(lines, id)
        if (visual != null && ocr != null) {
            val close = distance(visual.point, ocr.point) < .16f
            val boost = if (close) .08f else 0f
            return Target(visual.point, (visual.confidence + boost).coerceAtMost(.99f), "AVATAR+OCR")
        }
        // OCR alone is accepted only when the text is an exact alias and lies in
        // the middle of a plausible card row. This is still verified after tap.
        if (ocr != null && ocr.confidence >= .96f) return ocr
        return visual
    }

    fun scrollPoint(direction: Int = 1): Pair<Point, Point> =
        if (direction >= 0) Point(.52f, .80f) to Point(.52f, .30f)
        else Point(.52f, .30f) to Point(.52f, .80f)

    fun shouldScroll(now: Long): Boolean = now - lastScrollAt >= 850L
    fun markScroll(now: Long) { lastScrollAt = now }

    fun gridFingerprint(bitmap: Bitmap): Long {
        val small = Bitmap.createScaledBitmap(bitmap, 40, 24, true)
        val pixels = IntArray(40 * 24)
        small.getPixels(pixels, 0, 40, 0, 0, 40, 24)
        var h = 1469598103934665603L
        for (i in pixels.indices step 3) {
            h = h xor pixels[i].toLong()
            h *= 1099511628211L
        }
        if (small !== bitmap) small.recycle()
        lastGridFingerprint = h
        return h
    }

    private fun scanWholeGrid(src: Bitmap, template: Bitmap): Target? {
        val sw = 160
        val sh = max(74, (src.height * sw.toFloat() / src.width).toInt())
        val small = Bitmap.createScaledBitmap(src, sw, sh, true)
        val pixels = IntArray(sw * sh)
        small.getPixels(pixels, 0, sw, 0, 0, sw, sh)

        val tw = 32
        val th = 15
        val t = Bitmap.createScaledBitmap(template, tw, th, true)
        val tp = IntArray(tw * th)
        t.getPixels(tp, 0, tw, 0, 0, tw, th)

        var bestScore = -1f
        var bestX = -1
        var bestY = -1
        var second = -1f
        val cw = max(25, (sw * .178f).toInt())
        val ch = max(13, (sh * .175f).toInt())
        val step = 3
        val yMin = (sh * .10f).toInt().coerceAtLeast(0)
        val yMax = (sh * .90f).toInt().coerceAtMost(sh - ch - 1)

        var y = yMin
        while (y <= yMax) {
            var x = 0
            while (x <= sw - cw) {
                val score = patchScore(pixels, sw, sh, x, y, cw, ch, tp, tw, th)
                if (score > bestScore) { second = bestScore; bestScore = score; bestX = x; bestY = y }
                else if (score > second) second = score
                x += step
            }
            y += step
        }
        if (small !== src) small.recycle()
        t.recycle()

        if (bestX < 0 || bestScore < .80f) return null
        val margin = (bestScore - second).coerceAtLeast(0f)
        val confidence = (bestScore * .78f + min(1f, margin * 4.5f) * .22f).coerceIn(0f, .99f)
        if (confidence < .82f) return null
        return Target(
            Point(((bestX + cw / 2f) / sw).coerceIn(.02f, .98f), ((bestY + ch * .52f) / sh).coerceIn(.10f, .92f)),
            confidence,
            "AVATAR_SCAN"
        )
    }

    private fun scanExpectedGrid(src: Bitmap, template: Bitmap): Target? {
        // Fallback only after the adaptive scan. This keeps the known reference
        // layout useful on the supplied 1280x588 screenshots without making it
        // the primary coordinate system.
        val best = mutableListOf<Target>()
        for (row in 0 until 3) for (col in 0 until 5) {
            val card = cropReferenceGrid(src, col, row) ?: continue
            val score = similarity(template, card)
            card.recycle()
            if (score >= .82f) best += Target(cardCenter(col, row), score, "REFERENCE_GRID")
        }
        return best.maxByOrNull { it.confidence }
    }

    private fun loadTemplate(id: String): Bitmap? {
        if (cache.containsKey(id)) return cache[id]
        val b = try {
            assets.open("brawler_refs/$id.png").use { BitmapFactory.decodeStream(it) }
        } catch (_: Throwable) { null }
        cache[id] = b
        return b
    }

    private fun cropReferenceGrid(src: Bitmap, col: Int, row: Int): Bitmap? {
        val w = src.width; val h = src.height
        val left = ((.005f + col * .195f) * w).toInt().coerceAtLeast(0)
        val right = ((.182f + col * .195f) * w).toInt().coerceAtMost(w)
        val top = ((.162f + row * .272f) * h).toInt().coerceAtLeast(0)
        val bottom = ((.335f + row * .272f) * h).toInt().coerceAtMost(h)
        if (right - left < 20 || bottom - top < 15) return null
        return Bitmap.createBitmap(src, left, top, right - left, bottom - top)
    }

    private fun cardCenter(col: Int, row: Int) = Point(
        (.093f + col * .195f).coerceIn(.03f, .97f),
        (.248f + row * .272f).coerceIn(.10f, .90f)
    )

    private fun patchScore(src: IntArray, sw: Int, sh: Int, x: Int, y: Int, cw: Int, ch: Int,
                           tp: IntArray, tw: Int, th: Int): Float {
        var colorErr = 0.0
        var edgeErr = 0.0
        var n = 0
        var py = 0
        while (py < th) {
            val sy = y + (py * ch / th).coerceIn(0, ch - 1)
            var px = 0
            while (px < tw) {
                val sx = x + (px * cw / tw).coerceIn(0, cw - 1)
                val a = tp[py * tw + px]
                val b = src[sy * sw + sx]
                colorErr += (abs(Color.red(a) - Color.red(b)) + abs(Color.green(a) - Color.green(b)) + abs(Color.blue(a) - Color.blue(b))) / 765.0
                if (px > 0) {
                    val ap = tp[py * tw + px - 1]
                    val bp = src[sy * sw + (x + ((px - 1) * cw / tw).coerceIn(0, cw - 1))]
                    val ea = abs(Color.red(a)-Color.red(ap)) + abs(Color.green(a)-Color.green(ap)) + abs(Color.blue(a)-Color.blue(ap))
                    val eb = abs(Color.red(b)-Color.red(bp)) + abs(Color.green(b)-Color.green(bp)) + abs(Color.blue(b)-Color.blue(bp))
                    edgeErr += abs(ea-eb) / 765.0
                }
                n++; px++
            }
            py++
        }
        val colorScore = 1.0 - colorErr / n
        val edgeScore = if (tw <= 1) 1.0 else 1.0 - edgeErr / ((tw - 1) * th)
        return (colorScore * .78 + edgeScore * .22).toFloat().coerceIn(0f, 1f)
    }

    private fun similarity(a: Bitmap, b: Bitmap): Float {
        val aw = 32; val ah = 15
        val ar = Bitmap.createScaledBitmap(a, aw, ah, true)
        val br = Bitmap.createScaledBitmap(b, aw, ah, true)
        val ap = IntArray(aw * ah); val bp = IntArray(aw * ah)
        ar.getPixels(ap, 0, aw, 0, 0, aw, ah); br.getPixels(bp, 0, aw, 0, 0, aw, ah)
        var err = 0.0
        for (i in ap.indices) err += (abs(Color.red(ap[i])-Color.red(bp[i])) + abs(Color.green(ap[i])-Color.green(bp[i])) + abs(Color.blue(ap[i])-Color.blue(bp[i]))) / 765.0
        ar.recycle(); br.recycle()
        return (1.0 - err / ap.size).toFloat().coerceIn(0f,1f)
    }

    private fun scoreTextLine(line: OcrLine, aliases: List<String>): Float {
        val v = norm(line.text)
        val exact = if (aliases.any { it == v }) .98f else .90f
        val centerBonus = if (line.rect.center().x in .02f.. .98f) .02f else 0f
        return (exact + centerBonus).coerceAtMost(.99f)
    }

    private fun norm(s: String) = s.lowercase().replace('ё','е').replace(Regex("[^a-zа-я0-9]"), "")
    private fun distance(a: Point, b: Point): Float {
        val dx = a.x - b.x; val dy = a.y - b.y
        return kotlin.math.sqrt(dx * dx + dy * dy)
    }
}

class CupReader {
    fun read(lines: List<OcrLine>, now: Long): Int? {
        val fresh = lines.filter { now - it.capturedAt <= 2000L }
        val candidates = fresh.flatMap { line ->
            Regex("""(?<!\d)(\d{2,4})(?!\d)""").findAll(line.text)
                .mapNotNull { it.groupValues[1].toIntOrNull() }
                .filter { it in 0..10000 }
                .toList()
        }
        return candidates.firstOrNull()
    }

package com.rostik.touchbot

class CupTracker {
    private var lastValue: Int? = null
    private var confirmations = 0
    private var lastTimestamp = 0L

    fun offer(value: Int?, timestamp: Long): Int? {
        if (value == null || timestamp <= lastTimestamp) return null
        lastTimestamp = timestamp
        if (lastValue == value) confirmations++ else {
            lastValue = value
            confirmations = 1
        }
        return if (confirmations >= 3) value else null
    }

    fun reset() {
        lastValue = null
        confirmations = 0
        lastTimestamp = 0L
    }
}

===== app/src/main/java/com/rostik/touchbot/MatchSupervisor.kt =====
package com.rostik.touchbot

import android.graphics.Bitmap
import android.util.Log
import kotlinx.coroutines.*
import kotlinx.coroutines.channels.Channel
import java.util.concurrent.atomic.AtomicLong
import java.util.concurrent.atomic.AtomicReference

/**
 * Single owner of the navigation state machine.
 *
 * Critical actions are always perception -> tap -> verification. If the current
 * screen is not recognized with enough confidence, the supervisor waits instead
 * of tapping a guessed coordinate.
 */
class MatchSupervisor(private val queue: QueueEngine, private val assets: android.content.res.AssetManager): AutoCloseable {
    private val vision = MatchVision()
    private val ocr = OcrEngine()
    private val cupTracker = CupTracker()
    private val trophyReader = TrophyReader()
    private val navigation = NavigationVision()
    private val references = ReferenceMatcher(assets)
    private val selector = BrawlerSelector(assets)
    private val controller = ReactiveBrawlBallController()
    private val reward = RewardHandler()

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default)
    private val frameChannel = Channel<BitmapRef>(Channel.CONFLATED)
    private val latestScene = AtomicReference<MatchScene?>(null)
    private val latestOcr = AtomicReference<List<OcrLine>>(emptyList())
    private val latestOcrAt = AtomicLong(0L)

    @Volatile private var state = RunState.IDLE
    @Volatile private var running = false
    @Volatile private var paused = false
    @Volatile private var stateBeforePause = RunState.IDLE

    private var lastOcrAt = 0L
    private var stateSince = 0L
    private var actionBusyUntil = 0L
    private var workerStarted = false
    private var brawlerAttempts = 0
    private var brawlerScrolls = 0
    private var brawlerScrollDirection = 1
    private var lastBrawlerGridFingerprint: Long? = null
    private var unchangedGridCount = 0
    private var cupMismatchSince = 0L
    private var loadingEvidence = 0
    private var lastLoadingEvidenceAt = 0L
    private var matchStartedAt = 0L
    private var matchStartCups: Int? = null
    private var pendingDelta: Int? = null
    private var deltaConfirmations = 0
    private var matchResultRecorded = false
    private var resultSince = 0L
    private var rewardSince = 0L
    private var lastOutcome: MatchOutcome? = null
    private var advanceAfterCupVerification = false

    data class BitmapRef(val bitmap: Bitmap)

    fun start() {
        if (running) return
        running = true; paused = false
        transition(RunState.PREPARE)
        brawlerAttempts = 0; brawlerScrolls = 0; brawlerScrollDirection = 1
        lastBrawlerGridFingerprint = null; unchangedGridCount = 0
        cupMismatchSince = 0L; matchStartCups = null
        loadingEvidence = 0; lastLoadingEvidenceAt = 0L
        pendingDelta = null; deltaConfirmations = 0
        matchResultRecorded = false; resultSince = 0L; rewardSince = 0L; advanceAfterCupVerification = false
        cupTracker.reset(); reward.reset()
        StatsStore.beginRun()
        startWorkerIfNeeded()
    }

    fun pause() {
        if (running && !paused) { stateBeforePause = state; paused = true; transition(RunState.PAUSED) }
    }

    fun resume() {
        if (running && paused) { paused = false; transition(stateBeforePause) }
    }

    fun stop() {
        running = false; paused = false; transition(RunState.IDLE)
        StatsStore.endRun()
        drainFrames()
        latestScene.set(null); latestOcr.set(emptyList()); latestOcrAt.set(0L)
    }

    fun submitLatestFrame(bitmap: Bitmap) {
        if (!running) { safeRecycle(bitmap); return }
        val result = frameChannel.trySend(BitmapRef(bitmap))
        if (!result.isSuccess) safeRecycle(bitmap)
    }

    private fun startWorkerIfNeeded() {
        if (workerStarted) return
        workerStarted = true
        scope.launch {
            for (ref in frameChannel) {
                if (!running) { safeRecycle(ref.bitmap); continue }
                try { processFrame(ref.bitmap) }
                catch (t: Throwable) { Log.w("Supervisor", "frame failed", t) }
                finally { safeRecycle(ref.bitmap) }
            }
        }
    }

    private suspend fun processFrame(bitmap: Bitmap) {
        if (paused) return
        val now = System.currentTimeMillis()
        val scene = vision.detect(bitmap, now)
        latestScene.set(scene)

        val needOcr = when (state) {
            RunState.PLAYING -> now - lastOcrAt >= 900L
            RunState.RESULT, RunState.REWARD, RunState.PLAY_AGAIN -> now - lastOcrAt >= 160L
            else -> now - lastOcrAt >= 260L
        }
        if (needOcr) {
            lastOcrAt = now
            val lines = try { ocr.read(bitmap) } catch (t: Throwable) { Log.w("OCR", "failed", t); emptyList() }
            latestOcr.set(lines); latestOcrAt.set(now)
        }
        step(bitmap, scene, now)
    }

    private fun step(bitmap: Bitmap, scene: MatchScene, now: Long) {
        val obs = navigation.observe(freshOcr(now), scene)
        when (state) {
            RunState.PAUSED, RunState.IDLE, RunState.DONE, RunState.ERROR -> Unit

            RunState.PREPARE -> {
                if (queue.current() == null) transition(RunState.DONE) else transition(RunState.SELECT_BRAWLER)
            }

            RunState.SELECT_BRAWLER -> selectBrawler(bitmap, obs, now)
            RunState.VERIFY_BRAWLER -> verifyBrawler(obs, now)
            RunState.VERIFY_CUPS -> verifyCups(now)
            RunState.VERIFY_RESULT_CUPS -> verifyResultCups(obs, now)
            RunState.SELECT_MODE -> selectMode(bitmap, obs, now)
            RunState.SELECT_MAP -> selectMap(obs, now)
            RunState.PRESS_PLAY -> pressPlay(bitmap, obs, now)
            RunState.LOADING -> loading(scene, obs, now)
            RunState.PLAYING -> playing(scene, obs, now)
            RunState.RESULT -> result(obs, now)
            RunState.REWARD -> rewardStep(obs, now)
            RunState.PLAY_AGAIN -> playAgain(bitmap, obs, now)
        }

        if (state != RunState.PLAYING && state != RunState.IDLE && state != RunState.DONE && now - stateSince > stateTimeout(state)) {
            onTimeout(now)
        }
    }

    private fun selectBrawler(bitmap: Bitmap, obs: NavigationVision.Observation, now: Long) {
        val q = queue.current() ?: run { transition(RunState.DONE); return }
        val service = BotAccessibilityService.instance ?: return
        val lines = freshOcr(now)

        when (obs.type) {
            ScreenType.HOME -> {
                val target = navigation.findButton(lines, "brawlers", "бойцы", "fighters")
                if (target != null && now >= actionBusyUntil) {
                    service.tap(target)
                    actionBusyUntil = now + 850L
                    brawlerAttempts++
                    return
                }
                // Reference-authorized fallback: the tap is allowed only when the whole
                // HOME reference matches strongly. This is not a blind coordinate tap.
                if (now >= actionBusyUntil && references.score(bitmap, "home") >= .86f) {
                    if (service.tap(Point(.085f, .43f))) {
                        actionBusyUntil = now + 900L
                        brawlerAttempts++
                    }
                }
            }
            ScreenType.BRAWLER_LIST -> {
                val target = selector.find(bitmap, lines, q.brawlerId)
                if (target != null && now >= actionBusyUntil && target.confidence >= .84f) {
                    service.tap(target.point)
                    actionBusyUntil = now + 1000L
                    brawlerAttempts++
                    transition(RunState.VERIFY_BRAWLER)
                    return
                }

                if (now >= actionBusyUntil && now - stateSince > 1100L && brawlerScrolls < 30 && selector.shouldScroll(now)) {
                    val before = selector.gridFingerprint(bitmap)
                    if (lastBrawlerGridFingerprint != null) {
                        if (before == lastBrawlerGridFingerprint) unchangedGridCount++ else unchangedGridCount = 0
                        if (unchangedGridCount >= 2) brawlerScrollDirection *= -1
                    }
                    lastBrawlerGridFingerprint = before
                    val (from, to) = selector.scrollPoint(brawlerScrollDirection)
                    if (service.swipe(from, to)) {
                        selector.markScroll(now)
                        actionBusyUntil = now + 900L
                        brawlerScrolls++
                    }
                }
            }
            ScreenType.BRAWLER_PROFILE -> {
                val button = navigation.findButton(lines, "select", "выбрать", "choose")
                if (button != null && now >= actionBusyUntil) {
                    service.tap(button)
                    actionBusyUntil = now + 900L
                    transition(RunState.VERIFY_BRAWLER)
                }
            }
            else -> Unit
        }

        if (brawlerScrolls >= 30 && now - stateSince > 18_000L) {
            failSafe("Не удалось найти бойца ${BrawlerCatalog.name(q.brawlerId)} по аватарке/тексту")
        }
    }

    private fun verifyBrawler(obs: NavigationVision.Observation, now: Long) {
        val q = queue.current() ?: run { transition(RunState.DONE); return }
        val lines = freshOcr(now)
        val nameSeen = lines.any { lineContainsBrawler(it.text, q.brawlerId) }
        val accepted = when (obs.type) {
            ScreenType.BRAWLER_PROFILE -> nameSeen || obs.confidence >= .90f
            ScreenType.BRAWL_BALL_LOBBY, ScreenType.HOME -> nameSeen
            else -> false
        }
        if (accepted && now - stateSince >= 500L) {
            transition(RunState.VERIFY_CUPS)
            cupMismatchSince = 0L
        } else if (now - stateSince > 3200L && brawlerAttempts < 3) {
            transition(RunState.SELECT_BRAWLER)
        }
    }

    private fun verifyCups(now: Long) {
        val q = queue.current() ?: run { transition(RunState.DONE); return }
        val actual = trophyReader.currentBrawlerCups(freshOcr(now), q.currentCups) ?: return
        val configured = q.currentCups
        if (kotlin.math.abs(actual - configured) <= 8) {
            queue.updateCups(actual)
            matchStartCups = actual
            transition(RunState.SELECT_MODE)
            return
        }
        if (cupMismatchSince == 0L) cupMismatchSince = now
        if (now - cupMismatchSince > 2500L) {
            failSafe("Кубки не совпали: введено $configured, на экране $actual. Бот остановлен, чтобы не ошибиться.")
        }
    }

    private fun selectMode(bitmap: Bitmap, obs: NavigationVision.Observation, now: Long) {
        val service = BotAccessibilityService.instance ?: return
        val lines = freshOcr(now)
        val lobbyRef = references.score(bitmap, "brawl_ball_lobby")
        if ((obs.type == ScreenType.BRAWL_BALL_LOBBY && obs.brawlBall) || lobbyRef >= .86f) {
            transition(RunState.PRESS_PLAY)
            return
        }
        if (obs.type == ScreenType.HOME) {
            val current = navigation.findCurrentMode(lines)
            if (current != null && now >= actionBusyUntil) {
                if (service.tap(current)) { actionBusyUntil = now + 1200L }
                return
            }
            if (now >= actionBusyUntil && references.score(bitmap, "home") >= .86f) {
                // In the supplied HOME reference this point is the current-mode card.
                if (service.tap(Point(.60f, .91f))) actionBusyUntil = now + 1200L
            }
            return
        }
        if (obs.type != ScreenType.MODE_SELECT) return
        val ball = navigation.findBrawlBall(lines)
        if (ball != null && now >= actionBusyUntil) {
            if (service.tap(ball)) {
                actionBusyUntil = now + 1200L
                transition(RunState.SELECT_MAP)
            }
        } else if (now >= actionBusyUntil && references.score(bitmap, "mode_select") >= .86f) {
            if (service.tap(Point(.375f, .35f))) {
                actionBusyUntil = now + 1200L
                transition(RunState.SELECT_MAP)
            }
        }
    }

    private fun selectMap(obs: NavigationVision.Observation, now: Long) {
        if (obs.type == ScreenType.BRAWL_BALL_LOBBY && obs.brawlBall) transition(RunState.PRESS_PLAY)
    }

    private fun pressPlay(bitmap: Bitmap, obs: NavigationVision.Observation, now: Long) {
        val service = BotAccessibilityService.instance ?: return
        val lobbyRef = references.score(bitmap, "brawl_ball_lobby")
        if ((obs.type != ScreenType.BRAWL_BALL_LOBBY || !obs.brawlBall) && lobbyRef < .86f) return
        val p = obs.play ?: navigation.findButton(freshOcr(now), "play", "играть", "ready", "готов")
        val visualPlay = p ?: UiButtonVision.findYellowAction(bitmap)
        if (visualPlay != null && now >= actionBusyUntil) {
            if (service.tap(visualPlay)) {
                actionBusyUntil = now + 1500L
                matchStartCups = queue.current()?.currentCups
                matchStartedAt = now
                matchResultRecorded = false
                pendingDelta = null; deltaConfirmations = 0
                loadingEvidence = 0; lastLoadingEvidenceAt = 0L
                transition(RunState.LOADING)
            }
        }
    }

    private fun loading(scene: MatchScene, obs: NavigationVision.Observation, now: Long) {
        val evidence = obs.type == ScreenType.MATCH && scene.leftJoystick != null && scene.rightJoystick != null && scene.hudConfidence >= .42f && scene.player != null
        if (evidence) {
            if (now - lastLoadingEvidenceAt >= 180L) loadingEvidence++
            lastLoadingEvidenceAt = now
        }
        if (loadingEvidence >= 3) {
            matchStartedAt = now
            resultSince = 0L
            transition(RunState.PLAYING)
        }
    }

    private fun playing(scene: MatchScene, obs: NavigationVision.Observation, now: Long) {
        if (obs.type == ScreenType.RESULT || obs.type == ScreenType.REWARD) {
            transition(if (obs.type == ScreenType.REWARD) RunState.REWARD else RunState.RESULT)
            resultSince = now
            return
        }
        if (obs.type != ScreenType.MATCH) return

        val player = scene.player
        if (player == null) {
            // A player can disappear for a few frames because of effects/respawn.
            // Do not interpret that as match end.
            if (now - resultSince > 900L) resultSince = now
            return
        }
        resultSince = 0L
        val elapsed = now - matchStartedAt
        val frame = controller.next(scene, now, elapsed)
        if (now >= actionBusyUntil && scene.hudConfidence >= .42f) {
            val left = scene.leftJoystick ?: Point(.20f, .78f)
            val right = scene.rightJoystick ?: Point(.80f, .78f)
            if (BotAccessibilityService.instance?.dispatch(frame, left, right, 80L) == true) {
                actionBusyUntil = now + 70L
            }
        }
    }

    private fun result(obs: NavigationVision.Observation, now: Long) {
        if (obs.type != ScreenType.RESULT && obs.type != ScreenType.REWARD) return
        val lines = freshOcr(now)
        val delta = trophyReader.resultDelta(lines)
        if (delta != null) {
            if (pendingDelta == delta) deltaConfirmations++ else { pendingDelta = delta; deltaConfirmations = 1 }
            if (deltaConfirmations >= 3 && !matchResultRecorded) {
                applyConfirmedDelta(delta)
                matchResultRecorded = true
                lastOutcome = ResultReader.read(lines) ?: when {
                    delta > 0 -> MatchOutcome.WIN
                    delta < 0 -> MatchOutcome.LOSS
                    else -> MatchOutcome.DRAW
                }
                StatsStore.recordMatch(delta, lastOutcome!!)
            }
        }
        if (obs.type == ScreenType.REWARD) {
            if (matchResultRecorded) { transition(RunState.REWARD); rewardSince = now }
            return
        }
        if (matchResultRecorded && now - resultSince > 900L) transition(RunState.PLAY_AGAIN)
    }

    private fun rewardStep(obs: NavigationVision.Observation, now: Long) {
        val service = BotAccessibilityService.instance
        if (service != null && obs.rewardAction != null && now >= actionBusyUntil) {
            service.tap(obs.rewardAction.second); actionBusyUntil = now + 700L; rewardSince = now; return
        }
        if (now - rewardSince > 1200L) transition(RunState.PLAY_AGAIN)
    }

    private fun playAgain(bitmap: Bitmap, obs: NavigationVision.Observation, now: Long) {
        val service = BotAccessibilityService.instance ?: return
        val q = queue.current() ?: run { transition(RunState.DONE); return }
        if (!matchResultRecorded) return
        if (q.reached) {
            val proceed = obs.proceed ?: navigation.findButton(freshOcr(now), "proceed", "continue", "продолжить", "exit", "выйти")
            if (proceed != null && now >= actionBusyUntil && service.tap(proceed)) {
                actionBusyUntil = now + 1100L
                advanceAfterCupVerification = true
                transition(RunState.VERIFY_RESULT_CUPS)
            }
            return
        }
        val again = obs.playAgain ?: navigation.findButton(freshOcr(now), "play again", "играть снова", "ещё раз")
        val visualAgain = again ?: UiButtonVision.findYellowAction(bitmap)
        if (visualAgain != null && now >= actionBusyUntil && service.tap(visualAgain)) {
            actionBusyUntil = now + 1400L
            matchStartCups = q.currentCups
            matchStartedAt = now
            matchResultRecorded = false
            pendingDelta = null; deltaConfirmations = 0
            loadingEvidence = 0; lastLoadingEvidenceAt = 0L
            transition(RunState.LOADING)
        }
    }

    private fun verifyResultCups(obs: NavigationVision.Observation, now: Long) {
        val q = queue.current() ?: run { transition(RunState.DONE); return }
        val actual = trophyReader.currentBrawlerCups(freshOcr(now), q.currentCups) ?: return
        if (actual != q.currentCups) {
            if (now - stateSince > 3500L) failSafe("После результата кубки не подтверждены: ожидалось ${q.currentCups}, экран показывает $actual")
            return
        }
        if (advanceAfterCupVerification) {
            advanceAfterCupVerification = false
            if (queue.next()) { resetForNextBrawler(); transition(RunState.SELECT_BRAWLER) }
            else transition(RunState.DONE)
        } else transition(RunState.PLAY_AGAIN)
    }

    private fun applyConfirmedDelta(delta: Int) {
        val q = queue.current() ?: return
        val before = q.currentCups
        val after = before + delta
        if (after in 0..10000 && kotlin.math.abs(delta) <= 100) queue.updateCups(after)
    }

    private fun resetForNextBrawler() {
        brawlerAttempts = 0; brawlerScrolls = 0; brawlerScrollDirection = 1; unchangedGridCount = 0
        lastBrawlerGridFingerprint = null; cupMismatchSince = 0L
        matchStartCups = null; pendingDelta = null; deltaConfirmations = 0
        loadingEvidence = 0; lastLoadingEvidenceAt = 0L
        matchResultRecorded = false; resultSince = 0L; rewardSince = 0L; advanceAfterCupVerification = false; reward.reset()
    }

    private fun onTimeout(now: Long) {
        when (state) {
            RunState.SELECT_BRAWLER -> failSafe("Не удалось подтвердить выбор бойца")
            RunState.VERIFY_BRAWLER -> if (brawlerAttempts < 3) transition(RunState.SELECT_BRAWLER) else failSafe("Не подтверждён выбранный боец")
            RunState.VERIFY_CUPS -> failSafe("Не удалось надёжно прочитать кубки")
            RunState.SELECT_MODE, RunState.SELECT_MAP, RunState.PRESS_PLAY -> failSafe("Не удалось подтвердить Brawl Ball / PLAY")
            RunState.LOADING -> failSafe("Матч не перешёл в игровой экран")
            RunState.PLAYING -> failSafe("Игровой экран пропал без подтверждённого результата")
            RunState.RESULT, RunState.REWARD, RunState.VERIFY_RESULT_CUPS, RunState.PLAY_AGAIN -> failSafe("Результат матча не подтверждён")
            else -> Unit
        }
    }

    private fun transition(next: RunState) {
        state = next; stateSince = System.currentTimeMillis()
        Log.i("Supervisor", "state=$next")
    }

    private fun failSafe(reason: String) {
        Log.e("Supervisor", reason)
        running = false; paused = false; state = RunState.ERROR
        StatsStore.endRun()
    }

    fun status(): String {
        val q = queue.current(); val s = latestScene.get()
        val p = s?.player?.point?.let { "P=${(it.x*100).toInt()}%,${(it.y*100).toInt()}%" } ?: "P=--"
        val b = s?.ball?.point?.let { "B=${(it.x*100).toInt()}%,${(it.y*100).toInt()}%" } ?: "B=--"
        return "v30.2 | $state | ${q?.brawlerId ?: "-"} ${q?.currentCups ?: 0}/${q?.targetCups ?: 0} | $p $b"
    }

    private fun freshOcr(now: Long): List<OcrLine> = if (now - latestOcrAt.get() <= 2200L) latestOcr.get() else emptyList()
    private fun lineContainsBrawler(text: String, id: String): Boolean {
        val v = text.lowercase().replace('ё','е').replace(Regex("[^a-zа-я0-9]"), "")
        return BrawlerCatalog.aliases(id).any { a -> val n=a.lowercase().replace('ё','е').replace(Regex("[^a-zа-я0-9]"), ""); n.length >= 3 && (v.contains(n) || n.contains(v) && v.length >= 4) }
    }
    private fun stateTimeout(s: RunState) = when (s) {
        RunState.SELECT_BRAWLER -> 30_000L; RunState.VERIFY_BRAWLER -> 4_000L; RunState.VERIFY_CUPS -> 4_000L
        RunState.SELECT_MODE, RunState.SELECT_MAP, RunState.PRESS_PLAY -> 8_000L; RunState.LOADING -> 12_000L
        RunState.PLAYING -> 180_000L; RunState.RESULT -> 8_000L; RunState.REWARD -> 7_000L; RunState.VERIFY_RESULT_CUPS -> 5_000L; RunState.PLAY_AGAIN -> 7_000L
        else -> 30_000L
    }
    private fun drainFrames() { while (true) { val x = frameChannel.tryReceive().getOrNull() ?: break; safeRecycle(x.bitmap) } }
    private fun safeRecycle(b: Bitmap) { try { if (!b.isRecycled) b.recycle() } catch (_: Throwable) {} }
    override fun close() {
        running = false; paused = false; StatsStore.endRun()
        frameChannel.close(); scope.cancel(); vision.close(); ocr.close()
    }
}

package com.rostik.touchbot

import android.graphics.Bitmap
import android.graphics.Color
import kotlin.math.abs
import kotlin.math.hypot
import kotlin.math.max
import kotlin.math.min

/**
 * Adaptive marker detector.
 *
 * It deliberately supports both the green/cyan marker seen in some layouts
 * and blue/cyan team markers. The player's identity is not inferred from
 * color alone: spatial prior + indicator above + temporal continuity are used.
 */
class MatchVision {
    private var lastPlayer: Point? = null
    private var lastEnemy: Point? = null
    private var lastBall: Point? = null
    private var lastPlayerAt = 0L
    private var lastEnemyAt = 0L
    private var lastBallAt = 0L
    private var lastVelocity = Point(0f, 0f)

    fun detect(src: Bitmap, now: Long): MatchScene {
        val w = 360
        val h = 202
        val bmp = Bitmap.createScaledBitmap(src, w, h, true)
        val pixels = IntArray(w * h)
        bmp.getPixels(pixels, 0, w, 0, 0, w, h)

        val candidates = mutableListOf<Detection>()
        for (y in 25 until 180 step 2) {
            for (x in 8 until w - 8 step 2) {
                val c = pixels[y * w + x]
                val r = Color.red(c)
                val g = Color.green(c)
                val b = Color.blue(c)

                // Team marker: accept green/cyan/blue variants.
                val team = ((g > 150 && g > r * 1.28f && b > r * 1.10f) ||
                        (b > 160 && g > 125 && b > r * 1.35f && g > r * 1.18f))
                if (team) {
                    val p = Point(x / w.toFloat(), y / h.toFloat())
                    val score = markerScore(p, pixels, w, h, true)
                    if (score > 0.52f) candidates += Detection(p, score, "TEAM")
                }
            }
        }

        val grouped = cluster(candidates)
        val own = choosePlayer(grouped, pixels, w, h, now)
        val allies = grouped.filter { d -> own == null || distance(d.point, own.point) > .075f }
            .filter { it.confidence > .55f }
            .take(3)

        val enemies = findEnemies(pixels, w, h, now, own)
        val ball = findBall(pixels, w, h, now, own?.point)

        val leftJoy = findJoystick(pixels, w, h, 35, 145, 125, 198)
        val rightJoy = findJoystick(pixels, w, h, 235, 145, 355, 198)
        val hudConfidence = ((leftJoy?.confidence ?: 0f) + (rightJoy?.confidence ?: 0f)) / 2f
        val superPoint = findSuper(pixels, w, h)
        val confParts = listOfNotNull(own?.confidence, ball?.confidence, enemies.maxOfOrNull { it.confidence })
        val conf = if (confParts.isEmpty()) 0f else confParts.average().toFloat().coerceIn(0f, .98f)

        if (bmp !== src) try { bmp.recycle() } catch (_: Throwable) {}

        return MatchScene(
            player = own,
            enemies = enemies,
            allies = allies,
            ball = ball,
            superPoint = superPoint?.point,
            superReady = (superPoint?.confidence ?: 0f) >= .70f,
            leftJoystick = leftJoy?.point,
            rightJoystick = rightJoy?.point,
            hudConfidence = hudConfidence,
            confidence = conf,
            timestamp = now
        )
    }

    private fun choosePlayer(
        list: List<Detection>, pixels: IntArray, w: Int, h: Int, now: Long
    ): Detection? {
        if (list.isEmpty()) {
            if (lastPlayer != null && now - lastPlayerAt < 180L) {
                val p = Point(
                    (lastPlayer!!.x + lastVelocity.x * .08f).coerceIn(.04f, .96f),
                    (lastPlayer!!.y + lastVelocity.y * .08f).coerceIn(.12f, .92f)
                )
                return Detection(p, .52f, "PREDICTED_PLAYER")
            }
            return null
        }
        val scored = list.map { d ->
            val indicator = hasIndicatorAbove(d.point, pixels, w, h)
            val bottomPrior = 1f - (d.point.y - .55f).coerceAtLeast(0f) * 1.5f
            val temporal = lastPlayer?.let { (1f - distance(it, d.point) / .16f).coerceIn(0f, 1f) } ?: 0f
            d.copy(confidence = (d.confidence * .45f + indicator * .30f + bottomPrior.coerceIn(0f,1f)*.15f + temporal*.10f).coerceIn(0f,.99f))
        }.maxByOrNull { it.confidence }
        scored?.let {
            lastPlayer?.let { old ->
                val dt = ((now - lastPlayerAt).coerceAtLeast(1L)) / 1000f
                lastVelocity = Point(
                    ((it.point.x - old.x) / dt).coerceIn(-2f,2f),
                    ((it.point.y - old.y) / dt).coerceIn(-2f,2f)
                )
            }
            lastPlayer = it.point
            lastPlayerAt = now
        }
        return scored?.takeIf { it.confidence >= .52f }
    }

    private fun findEnemies(pix: IntArray, w: Int, h: Int, now: Long, own: Detection?): List<Detection> {
        val out = mutableListOf<Detection>()
        for (y in 28 until 155 step 2) for (x in 8 until w-8 step 2) {
            val c = pix[y*w+x]
            val r=Color.red(c); val g=Color.green(c); val b=Color.blue(c)
            val red = r > 155 && r > g*1.35f && r > b*1.25f && r-g > 55
            if (!red) continue
            val p=Point(x/w.toFloat(), y/h.toFloat())
            if (own != null && distance(p, own.point) < .10f) continue
            val temporal = lastEnemy?.let { (1f-distance(it,p)/.18f).coerceIn(0f,1f) } ?: 0f
            val cScore=(.58f + temporal*.22f + geometryScore(p)*.20f).coerceIn(0f,.98f)
            if (cScore > .52f) out += Detection(p,cScore,"ENEMY")
        }
        val grouped=cluster(out)
        val best=grouped.maxByOrNull { it.confidence }
        if (best != null) { lastEnemy=best.point; lastEnemyAt=now }
        return grouped.sortedByDescending { it.confidence }.take(3)
    }

    private fun findBall(pix:IntArray,w:Int,h:Int,now:Long,player:Point?):Detection? {
        var best:Detection?=null
        for(y in 35 until 160 step 2) for(x in 10 until w-10 step 2) {
            val c=pix[y*w+x]; val r=Color.red(c); val g=Color.green(c); val b=Color.blue(c)
            val bright=(r+g+b)/3
            val neutral=abs(r-g)<28 && abs(g-b)<28 && bright>135
            if(!neutral) continue
            val p=Point(x/w.toFloat(),y/h.toFloat())
            if(player!=null && distance(p,player)<.065f) continue
            val darkAround = neighborhoodDark(pix,w,h,x,y)
            val temporal=lastBall?.let{(1f-distance(it,p)/.16f).coerceIn(0f,1f)}?:0f
            val score=(.42f + darkAround*.25f + temporal*.25f + centerPrior(p)*.08f).coerceIn(0f,.98f)
            if(best==null || score>best!!.confidence) best=Detection(p,score,"BALL")
        }
        if(best!=null && best!!.confidence>.55f){ lastBall=best!!.point; lastBallAt=now; return best }
        if(lastBall!=null && now-lastBallAt<180L) return Detection(lastBall!!,.50f,"PREDICTED_BALL")
        return null
    }

    private fun findJoystick(pix:IntArray, w:Int, h:Int, x0:Int, y0:Int, x1:Int, y1:Int):Detection? {
        var best: Detection? = null
        val step = 4
        for (y in y0 until y1 step step) for (x in x0 until x1 step step) {
            val center = pix[y*w+x]
            val cb = (Color.red(center)+Color.green(center)+Color.blue(center))/3
            if (cb < 45) continue
            var ring = 0f
            var samples = 0
            val radius = 14
            for (a in 0 until 360 step 45) {
                val rad = Math.toRadians(a.toDouble())
                val xx=(x + kotlin.math.cos(rad)*radius).toInt()
                val yy=(y + kotlin.math.sin(rad)*radius).toInt()
                if(xx !in 0 until w || yy !in 0 until h) continue
                val c=pix[yy*w+xx]
                val b=(Color.red(c)+Color.green(c)+Color.blue(c))/3
                val contrast=abs(b-cb)/255f
                ring += contrast
                samples++
            }
            if(samples==0) continue
            val score=(ring/samples).coerceIn(0f,1f)
            if(score>.28f && (best==null || score>best!!.confidence)) {
                best=Detection(Point(x/w.toFloat(),y/h.toFloat()),score,"JOYSTICK")
            }
        }
        return best
    }

    private fun findSuper(pix:IntArray,w:Int,h:Int):Detection? {
        var best:Detection?=null
        for(y in 135 until 192 step 2) for(x in 245 until 355 step 2) {
            val c=pix[y*w+x]; val r=Color.red(c); val g=Color.green(c); val b=Color.blue(c)
            val bright=(r+g+b)/765f
            val colored=(b>120 && r>70 && bright>.45f)
            if(colored){
                val p=Point(x/w.toFloat(),y/h.toFloat())
                val score=(bright*.75f+.25f).coerceIn(0f,.98f)
                if(best==null || score>best!!.confidence) best=Detection(p,score,"SUPER")
            }
        }
        return best
    }

    private fun markerScore(p:Point,pix:IntArray,w:Int,h:Int,own:Boolean):Float {
        val above=hasIndicatorAbove(p,pix,w,h)
        val bottom=(1f-(p.y-.55f).coerceAtLeast(0f)*1.3f).coerceIn(0f,1f)
        return (.50f+above*.25f+bottom*.15f).coerceIn(0f,.95f)
    }

    private fun hasIndicatorAbove(p:Point,pix:IntArray,w:Int,h:Int):Float {
        val x=(p.x*w).toInt(); val y=(p.y*h).toInt()
        var bright=0
        for(dy in -14..-6) for(dx in -7..7){
            val xx=x+dx; val yy=y+dy
            if(xx !in 0 until w || yy !in 0 until h) continue
            val c=pix[yy*w+xx]
            if((Color.red(c)+Color.green(c)+Color.blue(c))/3>195) bright++
        }
        return (bright/80f).coerceIn(0f,1f)
    }

    private fun neighborhoodDark(p:IntArray,w:Int,h:Int,x:Int,y:Int):Float {
        var dark=0
        var n=0
        for(dy in -5..5) for(dx in -5..5){
            val xx=x+dx; val yy=y+dy
            if(xx !in 0 until w || yy !in 0 until h) continue
            n++; val c=p[yy*w+xx]
            if((Color.red(c)+Color.green(c)+Color.blue(c))/3<105) dark++
        }
        return if(n==0) 0f else dark/n.toFloat()
    }

    private fun geometryScore(p:Point)=((1f-abs(p.x-.5f)*1.2f).coerceIn(0f,1f))
    private fun centerPrior(p:Point)=((1f-distance(p,Point(.5f,.55f))/.7f).coerceIn(0f,1f))
    private fun distance(a:Point,b:Point)=hypot((a.x-b.x).toDouble(),(a.y-b.y).toDouble()).toFloat()

    private fun cluster(src:List<Detection>):List<Detection>{
        val out=mutableListOf<Detection>()
        for(d in src){
            val i=out.indexOfFirst{distance(it.point,d.point)<.035f}
            if(i<0) out+=d else if(d.confidence>out[i].confidence) out[i]=d
        }
        return out
    }

    fun close(){ lastPlayer=null; lastEnemy=null; lastBall=null }
}

package com.rostik.touchbot

import android.graphics.Bitmap
import java.util.concurrent.atomic.AtomicLong

data class Point(val x: Float, val y: Float)
data class RectN(val l: Float, val t: Float, val r: Float, val b: Float) {
    fun center() = Point((l + r) / 2f, (t + b) / 2f)
    fun area() = ((r-l).coerceAtLeast(0f) * (b-t).coerceAtLeast(0f))
}
data class Detection(val point: Point, val confidence: Float, val kind: String, val rect: RectN? = null)

enum class ScreenType { HOME, BRAWLER_LIST, BRAWLER_PROFILE, MODE_SELECT, BRAWL_BALL_LOBBY, LOADING, MATCH, RESULT, REWARD, UNKNOWN, PAUSED }

data class OcrLine(val text: String, val rect: RectN, val confidence: Float, val capturedAt: Long)

data class MatchScene(
    val player: Detection?,
    val enemies: List<Detection>,
    val allies: List<Detection>,
    val ball: Detection?,
    val superPoint: Point?,
    val superReady: Boolean,
    val leftJoystick: Point?,
    val rightJoystick: Point?,
    val hudConfidence: Float,
    val confidence: Float,
    val timestamp: Long
)

data class ControlFrame(
    val moveX: Float = 0f,
    val moveY: Float = 0f,
    val aimX: Float = 0f,
    val aimY: Float = 0f,
    val fire: Boolean = false,
    val useSuper: Boolean = false,
    val superPoint: Point? = null
)

data class QueueItem(
    val brawlerId: String,
    var currentCups: Int,
    var targetCups: Int,
    var startCups: Int = currentCups,
    var matchesPlayed: Int = 0
) {
    val remaining get() = (targetCups - currentCups).coerceAtLeast(0)
    val reached get() = currentCups >= targetCups
}

enum class RunState { IDLE, PREPARE, SELECT_BRAWLER, VERIFY_BRAWLER, VERIFY_CUPS, SELECT_MODE, SELECT_MAP, PRESS_PLAY, LOADING, PLAYING, RESULT, REWARD, VERIFY_RESULT_CUPS, PLAY_AGAIN, DONE, PAUSED, ERROR }

class FrameLease(private val bitmap: Bitmap) {
    private val refs = AtomicLong(1)
    fun retain(): FrameLease { refs.incrementAndGet(); return this }
    fun get(): Bitmap = bitmap
    fun release() {
        if (refs.decrementAndGet() == 0L && !bitmap.isRecycled) {
            try { bitmap.recycle() } catch (_: Throwable) {}
        }

package com.rostik.touchbot

/**
 * Conservative screen classifier. It prefers explicit UI anchors and the game HUD
 * over generic words. Unknown screens remain UNKNOWN and therefore produce no tap.
 */
class NavigationVision {
    data class Observation(
        val type: ScreenType,
        val confidence: Float,
        val brawlBall: Boolean,
        val play: Point? = null,
        val playAgain: Point? = null,
        val proceed: Point? = null,
        val rewardAction: Pair<String, Point>? = null
    )

    private var lastType = ScreenType.UNKNOWN
    private var stable = 0

    fun observe(lines: List<OcrLine>, scene: MatchScene?): Observation {
        val text = lines.joinToString(" ") { norm(it.text) }
        fun has(vararg words: String) = words.any { text.contains(norm(it)) }

        val brawlBall = has("brawl ball", "brawlball", "броулбол", "бравлболл")
        val result = has("victory", "победа", "defeat", "поражение", "defeat", "victory!")
        val reward = has("reward", "rewards", "награда", "claim", "получить", "open", "открыть", "daily wins")
        val brawlerHeading = has("brawlers", "бойцы", "fighters")
        val brawlerNameCount = lines.count { isLikelyBrawlerName(it.text) }
        val brawlerList = brawlerHeading || brawlerNameCount >= 3

        val modeHeading = has("trophy game modes", "game modes", "режимы игр", "режимы")
        val modeCards = has("knockout", "showdown", "gem grab", "hot zone", "wipeout", "brawl hockey")
        val modeSelect = (modeHeading || modeCards) && !brawlerList && !result

        val play = findButton(lines, "play", "играть", "ready", "готов", "play!")
        val again = findButton(lines, "play again", "играть снова", "ещё раз")
        val proceed = findButton(lines, "proceed", "continue", "продолжить", "exit", "выйти")
        val rewardAction = findButton(lines, "claim", "получить", "open", "открыть", "continue", "далее", "skip", "пропустить")
            ?.let { "REWARD" to it }

        val hud = scene?.let {
            it.leftJoystick != null && it.rightJoystick != null && it.hudConfidence >= .42f && it.player != null
        } ?: false
        val match = hud || (scene?.player != null && scene.hudConfidence >= .34f && !result && !brawlerList && !modeSelect)

        val candidate = when {
            result || again != null -> ScreenType.RESULT
            reward && !brawlerList && !modeSelect && !match -> ScreenType.REWARD
            match -> ScreenType.MATCH
            brawlBall && play != null -> ScreenType.BRAWL_BALL_LOBBY
            brawlerList && !modeSelect -> ScreenType.BRAWLER_LIST
            modeSelect -> ScreenType.MODE_SELECT
            play != null && brawlBall -> ScreenType.BRAWL_BALL_LOBBY
            has("shop", "магазин", "quests", "задания", "friends", "друзья") -> ScreenType.HOME
            play != null -> ScreenType.HOME
            else -> ScreenType.UNKNOWN
        }

        stable = if (candidate == lastType) stable + 1 else 1
        lastType = candidate
        val confidence = when {
            stable >= 3 -> .97f
            stable == 2 -> .89f
            else -> .72f
        }
        return Observation(candidate, confidence, brawlBall, play, again, proceed, rewardAction)
    }

    fun findButton(lines: List<OcrLine>, vararg words: String): Point? {
        val wanted = words.map(::norm).filter { it.length >= 3 }
        return lines.asSequence()
            .filter { it.rect.center().y > .25f }
            .mapNotNull { line ->
                val v = norm(line.text)
                val exact = wanted.any { w -> v == w || v.contains(w) || (w.contains(v) && v.length >= 4) }
                if (!exact) null else line
            }
            .maxByOrNull { buttonScore(it) }
            ?.rect?.center()
    }


    fun findCurrentMode(lines: List<OcrLine>): Point? {
        val modes = listOf(
            "solo showdown", "showdown", "knockout", "gem grab", "hot zone",
            "wipeout", "brawl hockey", "basket brawl", "heist"
        ).map(::norm)
        return lines.asSequence()
            .mapNotNull { line ->
                val v=norm(line.text)
                if (modes.any { m -> v == m || v.contains(m) }) line else null
            }
            .filter { it.rect.center().y > .72f }
            .maxByOrNull { 1f - kotlin.math.abs(it.rect.center().x - .55f) }
            ?.rect?.center()
    }
    fun findBrawlBall(lines: List<OcrLine>): Point? {
        val candidates = lines.filter { line ->
            val v = norm(line.text)
            v.contains("brawlball") || v.contains("броулбол") || v.contains("бравлболл")
        }
        return candidates.maxByOrNull { line ->
            val c = line.rect.center()
            // The supplied mode screen has Brawl Ball in the upper/middle card area.
            (1f - kotlin.math.abs(c.x - .50f)).coerceIn(0f, 1f) * .4f +
                (1f - kotlin.math.abs(c.y - .35f)).coerceIn(0f, 1f) * .6f
        }?.rect?.center()
    }

    private fun buttonScore(line: OcrLine): Float {
        val c = line.rect.center()
        val bottom = if (c.y > .55f) 1f else .5f
        val center = (1f - kotlin.math.abs(c.x - .5f)).coerceIn(0f,1f)
        return bottom * .65f + center * .35f
    }

    private fun isLikelyBrawlerName(text: String): Boolean {
        val v = norm(text)
        if (v.length < 3) return false
        return BrawlerCatalog.all.any { e -> BrawlerCatalog.aliases(e.id).any { norm(it) == v || (v.length >= 4 && v.contains(norm(it))) } }
    }

    private fun norm(s: String) = s.lowercase().replace('ё','е').replace(Regex("[^a-zа-я0-9]"), "")
}

===== app/src/main/java/com/rostik/touchbot/OcrEngine.kt =====
package com.rostik.touchbot

import android.graphics.Bitmap
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.latin.TextRecognizerOptions
import kotlinx.coroutines.suspendCancellableCoroutine
import kotlin.coroutines.resume
import kotlin.coroutines.resumeWithException

class OcrEngine {
    private val recognizer = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)

    suspend fun read(bitmap: Bitmap): List<OcrLine> =
        suspendCancellableCoroutine { cont ->
            val image = InputImage.fromBitmap(bitmap, 0)
            recognizer.process(image)
                .addOnSuccessListener { result ->
                    val now = System.currentTimeMillis()
                    val lines = result.textBlocks.flatMap { block ->
                        block.lines.map { line ->
                            val r = line.boundingBox
                            val w = bitmap.width.coerceAtLeast(1)
                            val h = bitmap.height.coerceAtLeast(1)
                            OcrLine(
                                line.text,
                                RectN(
                                    (r?.left ?: 0) / w.toFloat(),
                                    (r?.top ?: 0) / h.toFloat(),
                                    (r?.right ?: 0) / w.toFloat(),
                                    (r?.bottom ?: 0) / h.toFloat()
                                ),
                                0.8f,
                                now
                            )
                        }
                    }
                    if (cont.isActive) cont.resume(lines)
                }
                .addOnFailureListener { if (cont.isActive) cont.resumeWithException(it) }
        }

    fun close() { recognizer.close() }
}

===== app/src/main/java/com/rostik/touchbot/QueueEngine.kt =====
package com.rostik.touchbot

class QueueEngine {
    private val lock = Any()
    private val items = mutableListOf<QueueItem>()
    private var index = 0
    private var version = 0L

    fun replace(newItems: List<QueueItem>) = synchronized(lock) {
        items.clear(); items.addAll(newItems.map { it.copy() }); index = 0; version++
    }

    fun replacePreservingProgress(newItems: List<QueueItem>) = synchronized(lock) {
        val oldById = items.associateBy { it.brawlerId }
        items.clear()
        items.addAll(newItems.map { incoming ->
            oldById[incoming.brawlerId]?.copy(
                currentCups = incoming.currentCups,
                targetCups = incoming.targetCups
            ) ?: incoming.copy()
        })
        index = index.coerceIn(0, (items.size - 1).coerceAtLeast(0))
        version++
    }

    fun snapshot(): List<QueueItem> = synchronized(lock) { items.map { it.copy() } }
    fun current(): QueueItem? = synchronized(lock) { items.getOrNull(index)?.copy() }
    fun currentVersion(): Long = synchronized(lock) { version }

    fun updateCups(value: Int): Boolean = synchronized(lock) {
        val item = items.getOrNull(index) ?: return false
        if (value < 0 || value > 10000) return false
        if (kotlin.math.abs(value - item.currentCups) > 120) return false
        item.currentCups = value
        version++
        true
    }

    fun markMatch() = synchronized(lock) {
        items.getOrNull(index)?.let { it.matchesPlayed++ }
        version++
    }

    fun next(): Boolean = synchronized(lock) {
        if (index + 1 < items.size) { index++; version++; true } else false
    }

    fun finished(): Boolean = synchronized(lock) {
        items.isNotEmpty() && index == items.lastIndex && items.all { it.reached }
    }
}

package com.rostik.touchbot

import kotlin.math.hypot
import kotlin.math.max
import kotlin.math.min
import kotlin.random.Random

class ReactiveBrawlBallController {
    private var lastSuperAt = 0L
    private var lastDodgeAt = 0L
    private var patrolSign = 1f
    private var patrolUntil = 0L

    fun next(scene: MatchScene, now: Long, matchElapsedMs: Long): ControlFrame {
        val p=scene.player ?: return ControlFrame(moveY=.25f)
        if(now>=patrolUntil){
            patrolSign = if(Random.nextBoolean()) 1f else -1f
            patrolUntil = now + 900L + Random.nextLong(0,700)
        }

        val enemy=scene.enemies.minByOrNull{distance(it.point,p.point)}
        val ball=scene.ball
        val enemyThreat=enemy!=null && distance(enemy.point,p.point)<.20f

        if(enemyThreat && now-lastDodgeAt>800L){
            val dx=-(enemy!!.point.y-p.point.y)
            val dy=(enemy.point.x-p.point.x)
            val mag=max(.001f,hypot(dx.toDouble(),dy.toDouble()).toFloat())
            lastDodgeAt=now
            return ControlFrame(moveX=(dx/mag*patrolSign).coerceIn(-1f,1f),moveY=(dy/mag*patrolSign).coerceIn(-1f,1f))
        }

        val ballClose=ball!=null && distance(ball.point,p.point)<.13f
        val enemyBlocks=enemy!=null && ball!=null && isBetween(p.point,ball.point,enemy.point)
        val upwardAlly=scene.allies.filter{it.point.y<.35f && ball!=null && it.point.y<ball.point.y}
            .maxByOrNull{it.confidence}

        val attackPoint=when{
            enemyBlocks -> enemy!!.point
            ballClose && upwardAlly!=null -> upwardAlly.point
            ballClose -> Point(.5f,.08f)
            enemy!=null && distance(enemy.point,p.point)<.32f -> enemy.point
            else -> null
        }

        val aim=attackPoint ?: Point(.5f,.08f)
        val ax=(aim.x-p.point.x).coerceIn(-1f,1f)
        val ay=(aim.y-p.point.y).coerceIn(-1f,1f)

        val defendX=(.5f-p.point.x)*1.3f + patrolSign*.22f
        val defendY=(.73f-p.point.y)*1.3f
        val moveX=if(ball!=null && distance(ball.point,p.point)<.22f) (ball.point.x-p.point.x)*1.8f else defendX
        val moveY=if(ball!=null && distance(ball.point,p.point)<.22f) (ball.point.y-p.point.y)*1.5f else defendY

        val fire=attackPoint!=null
        val opportunity=enemy!=null && (enemyBlocks || ballClose || distance(enemy.point,p.point)<.30f)
        val useSuper=scene.superReady && opportunity && matchElapsedMs>=20_000L && now-lastSuperAt>=1_800L
        if(useSuper) lastSuperAt=now

        return ControlFrame(
            moveX=moveX.coerceIn(-1f,1f),
            moveY=moveY.coerceIn(-1f,1f),
            aimX=ax, aimY=ay, fire=fire,
            useSuper=useSuper, superPoint=scene.superPoint
        )
    }

    private fun isBetween(a:Point,b:Point,c:Point):Boolean{
        val abx=b.x-a.x; val aby=b.y-a.y
        val acx=c.x-a.x; val acy=c.y-a.y
        val len2=abx*abx+aby*aby
        if(len2<.0001f) return false
        val t=((acx*abx+acy*aby)/len2)
        if(t !in .12f..1.05f) return false
        val px=a.x+abx*t; val py=a.y+aby*t
        return hypot((c.x-px).toDouble(),(c.y-py).toDouble()).toFloat()<.09f
    }

    private fun distance(a:Point,b:Point)=hypot((a.x-b.x).toDouble(),(a.y-b.y).toDouble()).toFloat()
}

package com.rostik.touchbot

import android.content.res.AssetManager
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.graphics.Color
import kotlin.math.abs

/** Whole-screen reference matcher used only to authorize a conservative fallback tap. */
class ReferenceMatcher(private val assets: AssetManager) {
    private val cache = HashMap<String, Bitmap?>()

    fun score(screen: Bitmap, referenceName: String): Float {
        val ref = load(referenceName) ?: return 0f
        val sw = 48; val sh = 22
        val a = Bitmap.createScaledBitmap(screen, sw, sh, true)
        val b = Bitmap.createScaledBitmap(ref, sw, sh, true)
        val ap = IntArray(sw * sh); val bp = IntArray(sw * sh)
        a.getPixels(ap, 0, sw, 0, 0, sw, sh)
        b.getPixels(bp, 0, sw, 0, 0, sw, sh)
        var err = 0.0
        for (i in ap.indices) {
            err += (abs(Color.red(ap[i])-Color.red(bp[i])) + abs(Color.green(ap[i])-Color.green(bp[i])) + abs(Color.blue(ap[i])-Color.blue(bp[i]))) / 765.0
        }
        a.recycle(); b.recycle()
        return (1.0 - err / ap.size).toFloat().coerceIn(0f,1f)
    }

    private fun load(name: String): Bitmap? {
        if (cache.containsKey(name)) return cache[name]
        val b = try { assets.open("reference/$name.png").use { BitmapFactory.decodeStream(it) } } catch (_: Throwable) { null }
        cache[name] = b
        return b
    }
}

package com.rostik.touchbot

import android.util.Log

/**
 * Safe reward/popup state machine.
 * It never taps a generic screen center. Every tap is tied to OCR aliases
 * and/or a strongly constrained button geometry.
 */
class RewardHandler {
    private var actionCount = 0
    private var lastTapAt = 0L
    private var stableAction = ""
    private var stableCount = 0

    private val aliases = mapOf(
        "OPEN" to listOf("open","открыть","открыть!","открыть"),
        "CLAIM" to listOf("claim","забрать","получить"),
        "CONTINUE" to listOf("continue","продолжить","далее","дальше"),
        "HOME" to listOf("home","главная","выйти"),
        "CLOSE" to listOf("close","закрыть"),
        "SKIP" to listOf("skip","пропустить")
    )

    fun reset(){ actionCount=0; stableAction=""; stableCount=0; lastTapAt=0L }

    fun next(lines:List<OcrLine>, now:Long): Pair<String,Point>? {
        if(actionCount>=8) return null
        if(now-lastTapAt<450L) return null

        val candidates=lines.mapNotNull { line ->
            val normalized=line.text.lowercase().replace("ё","е").trim()
            aliases.entries.firstNotNullOfOrNull { (action, words) ->
                if(words.any{normalized.contains(it)}) action to line.rect.center() else null
            }?.let { it.first to it.second }
        }.filter { (_,p) -> p.y>.48f && p.y<.96f }

        val best=candidates.maxByOrNull { (_,p) -> if(p.x in .20f.. .85f) 1f else .5f } ?: return null
        if(best.first==stableAction) stableCount++ else { stableAction=best.first; stableCount=1 }
        if(stableCount<2) return null

        actionCount++
        lastTapAt=now
        Log.i("RewardHandler","confirmed action=${best.first} point=${best.second}")
        return best
    }

    fun exhausted()=actionCount>=8
}

    }

package com.rostik.touchbot

import android.content.Context
import kotlin.math.roundToInt

/** Persistent runtime statistics used by the app's "Статистика" section. */
object StatsStore {
    private const val PREFS = "v30_stats"
    private const val TOTAL_MS = "total_ms"
    private const val NET_CUPS = "net_cups"
    private const val WINS = "wins"
    private const val LOSSES = "losses"
    private const val DRAWS = "draws"
    private const val MATCHES = "matches"
    private const val ACTIVE_STARTED = "active_started"

    private lateinit var app: Context

    fun init(context: Context) { app = context.applicationContext }

    private fun prefs() = app.getSharedPreferences(PREFS, Context.MODE_PRIVATE)

    @Synchronized fun beginRun(now: Long = System.currentTimeMillis()) {
        if (prefs().getLong(ACTIVE_STARTED, 0L) == 0L) {
            prefs().edit().putLong(ACTIVE_STARTED, now).apply()
        }
    }

    @Synchronized fun endRun(now: Long = System.currentTimeMillis()) {
        val started = prefs().getLong(ACTIVE_STARTED, 0L)
        if (started > 0L) {
            val add = (now - started).coerceAtLeast(0L)
            prefs().edit()
                .putLong(TOTAL_MS, prefs().getLong(TOTAL_MS, 0L) + add)
                .putLong(ACTIVE_STARTED, 0L)
                .apply()
        }
    }

    @Synchronized fun recordMatch(cupDelta: Int, outcome: MatchOutcome) {
        val p = prefs()
        val e = p.edit()
            .putLong(NET_CUPS, p.getLong(NET_CUPS, 0L) + cupDelta)
            .putInt(MATCHES, p.getInt(MATCHES, 0) + 1)
        when (outcome) {
            MatchOutcome.WIN -> e.putInt(WINS, p.getInt(WINS, 0) + 1)
            MatchOutcome.LOSS -> e.putInt(LOSSES, p.getInt(LOSSES, 0) + 1)
            MatchOutcome.DRAW -> e.putInt(DRAWS, p.getInt(DRAWS, 0) + 1)
        }
        e.apply()
    }

    fun snapshot(now: Long = System.currentTimeMillis()): StatsSnapshot {
        val p = prefs()
        val active = p.getLong(ACTIVE_STARTED, 0L)
        val activeMs = if (active > 0L) (now - active).coerceAtLeast(0L) else 0L
        val matches = p.getInt(MATCHES, 0)
        val wins = p.getInt(WINS, 0)
        val losses = p.getInt(LOSSES, 0)
        val decisive = wins + losses
        val winRate = if (decisive == 0) 0 else ((wins.toDouble() / decisive) * 100.0).roundToInt()
        return StatsSnapshot(
            totalTimeMs = p.getLong(TOTAL_MS, 0L) + activeMs,
            netCups = p.getLong(NET_CUPS, 0L).toInt(),
            matches = matches,
            wins = wins,
            losses = losses,
            draws = p.getInt(DRAWS, 0),
            winRate = winRate
        )
    }

    @Synchronized fun reset() {
        prefs().edit().clear().apply()
    }
}

enum class MatchOutcome { WIN, LOSS, DRAW }

data class StatsSnapshot(
    val totalTimeMs: Long,
    val netCups: Int,
    val matches: Int,
    val wins: Int,
    val losses: Int,
    val draws: Int,
    val winRate: Int
) {
    fun timeText(): String {
        val totalSeconds = totalTimeMs / 1000L
        val hours = totalSeconds / 3600L
        val minutes = (totalSeconds % 3600L) / 60L
        val seconds = totalSeconds % 60L
        return if (hours > 0) "%02d:%02d:%02d".format(hours, minutes, seconds)
        else "%02d:%02d".format(minutes, seconds)
    }
}

package com.rostik.touchbot

/** Conservative OCR reader for signed result deltas and current brawler trophies. */
class TrophyReader {
    fun resultDelta(lines: List<OcrLine>): Int? {
        val candidates = lines.asSequence()
            .filter { it.rect.t < .55f && it.rect.l < .75f }
            .flatMap { line ->
                Regex("""([+-])\s*(\d{1,3})""").findAll(line.text.replace('−','-')).mapNotNull {
                    val n = it.groupValues[2].toIntOrNull() ?: return@mapNotNull null
                    if (n <= 100) (if (it.groupValues[1] == "+") n else -n) else null
                }
            }
            .toList()
        return candidates.distinct().singleOrNull() ?: candidates.firstOrNull()
    }

    fun currentBrawlerCups(lines: List<OcrLine>, expected: Int? = null): Int? {
        data class Candidate(val value: Int, val line: OcrLine)
        val candidates = lines.asSequence()
            .flatMap { line ->
                Regex("(?<!\\d)(\\d{1,4})(?!\\d)").findAll(line.text).mapNotNull {
                    val n = it.groupValues[1].toIntOrNull() ?: return@mapNotNull null
                    if (n in 0..5000) Candidate(n, line) else null
                }
            }
            .filter { c ->
                val r = c.line.rect
                // Trophy counters in the supplied references live in the card/profile
                // content area, not in the top navigation bar or bottom tab bar.
                r.t < .86f && r.b > .08f
            }
            .toList()

        if (candidates.isEmpty()) return null
        val target = expected?.coerceIn(0,5000)
        return candidates.minByOrNull { c ->
            val r = c.line.rect
            val spatial = kotlin.math.abs(r.center().x - .25f) + kotlin.math.abs(r.center().y - .52f)
            val expectedPenalty = if (target == null) 0f else kotlin.math.abs(c.value - target) / 500f
            spatial + expectedPenalty
        }?.value
    }
}

===== app/src/main/java/com/rostik/touchbot/UiButtonVision.kt =====
package com.rostik.touchbot

import android.graphics.Bitmap
import android.graphics.Color
import kotlin.math.max

/** Lightweight visual fallback for the large yellow action buttons used by the supplied UI. */
object UiButtonVision {
    fun findYellowAction(bitmap: Bitmap): Point? {
        val w = 160
        val h = max(74, bitmap.height * w / bitmap.width)
        val small = Bitmap.createScaledBitmap(bitmap, w, h, true)
        val px = IntArray(w*h)
        small.getPixels(px,0,w,0,0,w,h)
        val visited = BooleanArray(w*h)
        var bestArea = 0
        var best: IntArray? = null
        fun yellow(c:Int):Boolean {
            val r=Color.red(c); val g=Color.green(c); val b=Color.blue(c)
            return r>175 && g>105 && r>g*1.15f && b<115
        }
        val qx=IntArray(w*h); val qy=IntArray(w*h)
        for(y in (h*0.68f).toInt() until h-2) for(x in 0 until w) {
            val idx=y*w+x
            if(visited[idx] || !yellow(px[idx])) continue
            var head=0; var tail=0; qx[tail]=x; qy[tail]=y; tail++; visited[idx]=true
            var minX=x; var maxX=x; var minY=y; var maxY=y; var area=0
            while(head<tail){
                val cx=qx[head]; val cy=qy[head]; head++; area++
                if(cx<minX)minX=cx; if(cx>maxX)maxX=cx; if(cy<minY)minY=cy; if(cy>maxY)maxY=cy
                val ns=intArrayOf((cy-1)*w+cx,(cy+1)*w+cx,cy*w+cx-1,cy*w+cx+1)
                for(ni in ns){
                    if(ni<0 || ni>=px.size) continue
                    val nx=ni%w; val ny=ni/w
                    if(ny<(h*0.68f).toInt() || nx<0 || nx>=w || visited[ni] || !yellow(px[ni])) continue
                    visited[ni]=true; qx[tail]=nx; qy[tail]=ny; tail++
                }
            }
            if(area>bestArea && maxX-minX>12 && maxY-minY>5) { bestArea=area; best=intArrayOf(minX,minY,maxX,maxY) }
        }
        if (small !== bitmap) small.recycle()
        val r=best ?: return null
        return Point(((r[0]+r[2])/2f/w).coerceIn(0f,1f),((r[1]+r[3])/2f/h).coerceIn(0f,1f))
    }
}
